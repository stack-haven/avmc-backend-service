# 审计复现用例

把本文件的代码落盘为 `pkg/textenhance/audit_test.go` 即可复现评分报告中的全部结论。
用例设计为「观察式」（PASS + t.Log 输出证据），不会污染 CI；
**M1 里程碑应把它们改写为断言式失败用例，先红后绿。**

```bash
cd backend-service
go test ./pkg/textenhance/ -run TestAudit -v
go test -race ./pkg/textenhance/ -run TestAudit_ECRace
go test ./pkg/textenhance/ -run XXX -bench BenchmarkAudit -benchtime 20x
```

```go
package textenhance_test

import (
	"context"
	"sort"
	"sync"
	"testing"

	"backend-service/pkg/textenhance"
	"backend-service/pkg/textenhance/builtins"
	"backend-service/pkg/textenhance/processors"
)

func newBigVocab() *processors.VocabularySnapshot {
	entries := make([]*processors.VocabularyEntry, 0, 2000)
	rels := make([]*processors.VocabularyRelation, 0, 2000)
	for i := 0; i < 2000; i++ {
		entries = append(entries, &processors.VocabularyEntry{
			ID: uint32(i + 1), StandardText: string(rune('一'+i%2000)) + "术语项目", Category: "TERM",
		})
		rels = append(rels, &processors.VocabularyRelation{
			EntryID: uint32(i + 1), RelationType: "ALIAS",
			RelatedText: string(rune('あ'+i%1000)) + "别名", TargetEntryID: uint32(i + 1),
		})
	}
	return textenhance.NewVocabularySnapshot(entries, rels)
}

// C1：Pipeline.Run(ctx, nil) 的 panic 逃逸出 Run，HA 第一重保护失效。
func TestAudit_NilEC(t *testing.T) {
	defer func() {
		if r := recover(); r != nil {
			t.Logf("C1 FINDING: nil-ec panic escapes Run: %v", r)
		}
	}()
	p, _ := textenhance.BuildPipeline(builtins.NewDefaultRegistry(), textenhance.DefaultPolicy())
	p.Run(context.Background(), nil)
	t.Log("C1: no panic (已修复)")
}

// C2：Policy 阈值未透传给 processor（builder.go 调 reg.Build(name) 不带 opts）。
func TestAudit_PolicyThresholdsIgnored(t *testing.T) {
	pol := textenhance.DefaultPolicy()
	pol.FuzzyAutoThreshold = 0.99
	p, err := textenhance.BuildPipeline(builtins.NewDefaultRegistry(), pol)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("C2 FINDING: built %d procs, FuzzyAutoThreshold=0.99 未生效（工厂未收到 Option）", len(p.Processors()))
}

// C4：单句产出数千条 Change；C3/性能：逐 processor 耗时。
func TestAudit_ChangeExplosionAndTimings(t *testing.T) {
	p, _ := textenhance.BuildPipeline(builtins.NewDefaultRegistry(), textenhance.DefaultPolicy())
	ec := textenhance.NewEnhancementContext(
		"呃，这个项目里我们讨论了一术语项目和二术语项目的落地，另外还有三术语项日需要攻克技术难点。",
		newBigVocab(), nil)
	p.Run(context.Background(), ec)

	tm := ec.GetTimings()
	keys := make([]string, 0, len(tm))
	for k := range tm {
		keys = append(keys, k)
	}
	sort.Slice(keys, func(i, j int) bool { return tm[keys[i]] > tm[keys[j]] })
	for _, k := range keys {
		t.Logf("  %-28s %v", k, tm[k])
	}
	t.Logf("C4 FINDING: changes=%d lockedSpans=%d elapsed=%v", len(ec.GetChanges()), len(ec.LockedSpans), ec.Elapsed())
}

// C5：阈值越界被归零而非回落默认值 → fuzzy 变成无差别替换。
func TestAudit_ClampToZero(t *testing.T) {
	pol := &textenhance.Policy{EnabledProcessors: []string{"fuzzy_matching"}, FuzzyAutoThreshold: 1.5}
	v, errs := pol.Validate()
	t.Logf("C5 FINDING: 1.5 -> %v (errs=%v)", v.FuzzyAutoThreshold, errs)
}

// C6：空 EnabledProcessors 文档说「不启用任何」，实际还原为全量。
func TestAudit_EmptyPolicyEnablesAll(t *testing.T) {
	v, _ := (&textenhance.Policy{EnabledProcessors: []string{}}).Validate()
	t.Logf("C6 FINDING: empty -> %v", v.EnabledProcessors)
}

// C9：Lock 无去重、无界增长，IsLocked O(n)。
func TestAudit_LockedSpanGrowth(t *testing.T) {
	ec := textenhance.NewEnhancementContext("x", nil, nil)
	for i := 0; i < 5; i++ {
		ec.Lock("同一个词")
	}
	t.Logf("C9 FINDING: LockedSpans=%v", ec.LockedSpans)
}

// S1：EnhancementContext 宣称线程安全，但 processor 绕过 mutex 直接赋值 → 实测 race。
func TestAudit_ECRace(t *testing.T) {
	proc, _ := builtins.NewDefaultRegistry().Build("cleaning")
	ec := textenhance.NewEnhancementContext("a  b\x01c,,,d", nil, nil)
	var wg sync.WaitGroup
	wg.Add(2)
	go func() { defer wg.Done(); proc.Process(context.Background(), ec) }()
	go func() { defer wg.Done(); ec.AppendChange(processors.Change{}) }()
	wg.Wait()
}

// 性能基线：M6 里程碑的验收目标 < 200µs / < 32KB / < 200 allocs。
func BenchmarkAudit_Pipeline(b *testing.B) {
	v := newBigVocab()
	p, _ := textenhance.BuildPipeline(builtins.NewDefaultRegistry(), textenhance.DefaultPolicy())
	text := "呃，这个项目里我们讨论了一术语项目和二术语项目的落地，另外还有三术语项日需要攻克技术难点。"
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		ec := textenhance.NewEnhancementContext(text, v, nil)
		p.Run(context.Background(), ec)
	}
}
```
